cmake_minimum_required(VERSION 3.20)
project(cuda_manufacturing_portfolio LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_CUDA "Build and test CUDA reference (requires an allocated GPU for tests)" OFF)

add_executable(calibrate_cpu src/lab01/calibrate_cpu.cpp)
if(MSVC)
  target_compile_options(calibrate_cpu PRIVATE /W4)
else()
  target_compile_options(calibrate_cpu PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
foreach(n IN ITEMS 1 255 256 257 1000003)
  add_test(NAME cpu_${n} COMMAND calibrate_cpu ${n})
endforeach()
foreach(value IN ITEMS 0 -1 abc 12x 100000001)
  add_test(NAME cpu_invalid_${value} COMMAND calibrate_cpu ${value})
  set_tests_properties(cpu_invalid_${value} PROPERTIES WILL_FAIL TRUE)
endforeach()

if(ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    message(FATAL_ERROR "Set CMAKE_CUDA_ARCHITECTURES explicitly (80 for the A100 lab).")
  endif()
  enable_language(CUDA)
  add_executable(calibrate_cuda src/lab01/calibrate_cuda.cu)
  set_target_properties(calibrate_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(calibrate_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  foreach(n IN ITEMS 1 255 256 257 1000003)
    add_test(NAME gpu_${n} COMMAND calibrate_cuda ${n})
    set_tests_properties(gpu_${n} PROPERTIES LABELS gpu)
  endforeach()
endif()
